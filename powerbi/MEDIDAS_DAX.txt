// ========================================
// MEDIDAS DAX PARA POWER BI
// Copie e cole no Power BI Desktop
// ========================================

// MEDIDA 1: Total de Equipamentos
Total Equipamentos = COUNT(vw_powerbi_equipamentos[Número de Série])

// MEDIDA 2: Equipamentos Críticos
Equipamentos Críticos = 
CALCULATE(
    COUNT(vw_powerbi_equipamentos[Número de Série]),
    vw_powerbi_equipamentos[Dias em Estoque] > 14
)

// MEDIDA 3: Média de Dias
Média Dias = 
AVERAGE(vw_powerbi_equipamentos[Dias em Estoque])

// MEDIDA 4: % Crítico
% Crítico = 
DIVIDE(
    [Equipamentos Críticos],
    [Total Equipamentos],
    0
) * 100

// MEDIDA 5: Equipamentos 0-7 Dias
0-7 Dias = 
CALCULATE(
    COUNT(vw_powerbi_equipamentos[Número de Série]),
    vw_powerbi_equipamentos[Dias em Estoque] >= 0,
    vw_powerbi_equipamentos[Dias em Estoque] <= 7
)

// MEDIDA 6: Equipamentos 7-14 Dias
7-14 Dias = 
CALCULATE(
    COUNT(vw_powerbi_equipamentos[Número de Série]),
    vw_powerbi_equipamentos[Dias em Estoque] > 7,
    vw_powerbi_equipamentos[Dias em Estoque] <= 14
)

// MEDIDA 7: Equipamentos Acima 14 Dias
Acima 14 Dias = 
CALCULATE(
    COUNT(vw_powerbi_equipamentos[Número de Série]),
    vw_powerbi_equipamentos[Dias em Estoque] > 14
)

// MEDIDA 8: Total de Técnicos
Total Técnicos = 
DISTINCTCOUNT(vw_powerbi_equipamentos[Nome do Técnico])

// MEDIDA 9: Total de Supervisores
Total Supervisores = 
DISTINCTCOUNT(vw_powerbi_equipamentos[Supervisor])
